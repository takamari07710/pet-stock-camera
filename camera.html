function setRemainder( jan, remainder ) { jan = String(jan).trim(); remainder = Number(remainder); /* * 整数チェック */ if ( !Number.isInteger(remainder) ) { throw new Error( '端数は整数で入力してください。' ); } /* * マイナスだけ禁止 */ if (remainder < 0) { throw new Error( '端数は0以上で入力してください。' ); } const sheet = getStockSheet(); const values = sheet .getDataRange() .getValues(); /* * A列 = JANコード * B列 = 商品名 * C列 = パック数 * D列 = 1パックあたり * E列 = 端数 * F列 = 合計 */ for ( let i = 1; i < values.length; i++ ) { const code = String( values[i][0] ).trim(); if (code !== jan) { continue; } const row = i + 1; const name = values[i][1]; const packCount = Number( values[i][2] ) || 0; const perPack = Number( values[i][3] ) || 0; /* * E列に端数を保存 */ sheet .getRange(row, 5) .setValue( remainder ); /* * F列の合計を計算 * * パック数 × 1パックあたり + 端数 */ const total = packCount * perPack + remainder; /* * F列に合計を保存 */ sheet .getRange(row, 6) .setValue( total ); /* * 履歴 */ const historySheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName( HISTORY_SHEET_NAME ); if (historySheet) { historySheet.appendRow([ new Date(), jan, name, '端数変更', remainder, total ]); } /* * 結果 */ return { success: true, code: jan, name: name, packCount: packCount, perPack: perPack, remainder: remainder, total: total }; } return { success: false, notFound: true, message: '登録されていないJANコードです。', code: jan }; }
